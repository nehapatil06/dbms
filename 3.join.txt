CREATE TABLE Student2 (
    student_id INT PRIMARY KEY,
    name VARCHAR2(10),
    age INT,
    dept VARCHAR2(10)
);

CREATE TABLE Enrollment2 (
    enroll_id INT PRIMARY KEY,
    student_id INT,
    course_name VARCHAR2(10),
    marks INT,
    FOREIGN KEY (student_id) REFERENCES Student2(student_id)
);

SELECT s.student_id, s.name, e.course_name, e.marks
FROM Student2 s
INNER JOIN Enrollment2 e
ON s.student_id = e.student_id;


SELECT s.name, e.course_name, e.marks
FROM Student2 s
JOIN Enrollment2 e
ON s.student_id = e.student_id
WHERE e.marks > 80;


SELECT MAX(marks) AS Highest_SQL_Marks
FROM Enrollment2
WHERE course_name = 'SQL';


SELECT s.dept, AVG(e.marks) AS Avg_Marks
FROM Student2 s
JOIN Enrollment2 e
ON s.student_id = e.student_id
GROUP BY s.dept;
