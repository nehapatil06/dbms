STEP 4: Create Unnamed PL/SQL code block to control structure and do exception handling .
SQL> DECLARE
rno NUMBER(3) := &roll_no;
bname VARCHAR2(12) := '&n_of_book';
no_days NUMBER(7);
issuedate DATE;
fineamt NUMBER(5) := 0;
BEGIN
SELECT d_of_issue INTO issuedate FROM brw WHERE roll_no= rno;
SELECT sysdate - to_date(issuedate) days INTO no_days FROM dual;
IF (no_days >=15 AND no_days<=30) THEN
fineamt:= no_days*5;
ELSIF no_days > 30 THEN
fineamt := no_days*50;
END IF;
UPDATE brw SET status ='R' WHERE roll_no=rno;
IF fineamt>=0 THEN
INSERT INTO br_fine VALUES(rno,sysdate,fineamt);
END IF;
EXCEPTION
WHEN no_data_found THEN
dbms_output.put_line('Record not found');
END;
